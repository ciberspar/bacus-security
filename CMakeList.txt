
cmake_minimum_required(VERSION 3.16)

project(BacusProtector
    VERSION 0.1.0
    DESCRIPTION "Bacus Protector security application"
    LANGUAGES CXX
)

option(BACUS_BUILD_TUI "Build terminal interface" ON)
option(BACUS_BUILD_GUI "Build graphical interface" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(bacus_core
    src/core/protector.cpp
)

target_include_directories(bacus_core
    PUBLIC ${PROJECT_SOURCE_DIR}/include
)

if(BACUS_BUILD_TUI)
    add_executable(bacus-protector-tui
        src/main.cpp
        src/tui/tui.cpp
    )
    target_link_libraries(bacus-protector-tui PRIVATE bacus_core)
endif()

if(BACUS_BUILD_GUI)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)

    qt_standard_project_setup()

    qt_add_executable(bacus-protector-gui
        src/gui/gui.cpp
    )

    target_link_libraries(bacus-protector-gui
        PRIVATE bacus_core Qt6::Widgets
    )
endif()

if(MSVC)
    target_compile_options(bacus_core PRIVATE /W4)
else()
    target_compile_options(bacus_core PRIVATE
        -Wall -Wextra -Wpedantic
    )
endif()
