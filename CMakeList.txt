
cmake_minimum_required(VERSION 3.16)

project(BacusProtector
    VERSION 0.1.0
    DESCRIPTION "Bacus Protector security scanner"
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(bacus-protector
    src/main.cpp
)

target_include_directories(bacus-protector
    PRIVATE
        ${PROJECT_SOURCE_DIR}/include
)

if(MSVC)
    target_compile_options(bacus-protector PRIVATE /W4)
else()
    target_compile_options(bacus-protector PRIVATE
        -Wall
        -Wextra
        -Wpedantic
    )
endif()

enable_testing()
